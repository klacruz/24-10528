interface MetricCardProps {
  label: string;
  value: string;
  change: string;
  status: "positive" | "warning" | "danger";
}

function MetricCard({
  label,
  value,
  change,
  status
}: MetricCardProps) {
  return (
    <div className="metricCard">
      <span className="metricLabel">{label}</span>

      <div className="metricBottom">
        <strong>{value}</strong>

        <span className={`metricChange ${status}`}>
          {change}
        </span>
      </div>
    </div>
  );
}

function InfoRow({
  label,
  value
}: {
  label: string;
  value: string;
}) {
  return (
    <div className="infoRow">
      <span>{label}</span>
      <strong>{value}</strong>
    </div>
  );
}

function StatusItem({
  title,
  description,
  status
}: {
  title: string;
  description: string;
  status: "success" | "warning" | "danger";
}) {
  return (
    <div className="statusItem">
      <span className={`statusIndicator ${status}`} />

      <div>
        <strong>{title}</strong>
        <span>{description}</span>
      </div>
    </div>
  );
}

## 5. Para las imágenes: `contain` vs `cover`

En una aplicación de monitoreo, esto importa bastante.

Si la imagen contiene información y **no se puede recortar**, usa:

```css
img {
  width: 100%;
  height: 100%;
  object-fit: contain;
}
```

Así:

```text
┌─────────────────────┐
│                     │
│   ┌─────────────┐   │
│   │   IMAGEN    │   │
│   └─────────────┘   │
│                     │
└─────────────────────┘
```

La imagen completa siempre permanece visible.

Si es una fotografía puramente decorativa y prefieres llenar todo el espacio:

```css
img {
  width: 100%;
  height: 100%;
  object-fit: cover;
}
```

Pero puede recortar los extremos.

**Para Test usaría `contain` casi siempre.**

También puedes evitar deformaciones incluso sin altura fija:

```css
img {
  width: 100%;
  height: auto;
  display: block;
}
```

Nunca haría esto:

```css
/* ❌ Puede deformar la imagen */
img {
  width: 500px;
  height: 300px;
}
```

sin acompañarlo de `object-fit`.

La interfaz principal queda así:

```ts
export interface StatisticalModel {
  readonly name: string;

  process(
    microsensors: readonly number[]
  ): StatisticalResult;
}
```

Y el modelo concreto recibe el radio por constructor:

```ts
const statisticalModel =
  new ToleranceMeanModel(0.10);

const batchProcessor =
  new BatchProcessor(statisticalModel);
```

El cálculo implementado hace exactamente lo que pediste. Primero:

```ts
const initialMean =
  mean(microsensors);
```

Después calcula el margen a partir del promedio:

```ts
const toleranceMargin =
  Math.abs(initialMean) * this.toleranceRadius;
```

Y clasifica cada microsensor:

```ts
for (const value of microsensors) {
  const isInsideTolerance =
    Math.abs(value - initialMean) <= toleranceMargin;

  (isInsideTolerance
    ? cleanValues
    : outliers
  ).push(value);
}
```

Finalmente:

```ts
const finalMean =
  cleanValues.length > 0
    ? mean(cleanValues)
    : initialMean;
```

### 1. Cada batch ahora guarda cuándo llegó

Añadí `receivedAt` porque para una serie temporal es preferible trabajar con el momento de llegada real y no simplemente con `Batch 1`, `Batch 2`, etc.

```ts
export interface RawBatch {
  batchId: number;
  receivedAt: number;
  sensors: SensorReading[];
}

export interface ProcessedBatch {
  batchId: number;
  receivedAt: number;
  sensors: ProcessedSensorReading[];
}
```

Cuando llega un batch desde el worker:

```ts
if (parsed.kind === 'batch') {
  setRawBatches((batches) => {
    const next = [
      ...batches,
      {
        batchId: parsed.batchId,
        receivedAt: Date.now(),
        sensors: [],
      },
    ];

    return next.length > MAX_BATCHES
      ? next.slice(-MAX_BATCHES)
      : next;
  });

  return;
}
```

Y `BatchProcessor` simplemente conserva el timestamp:

```ts
process(batch: RawBatch): ProcessedBatch {
  return {
    batchId: batch.batchId,
    receivedAt: batch.receivedAt,

    sensors: batch.sensors.map((sensor) => ({
      sensorIndex: sensor.sensorIndex,
      sensorId: sensor.sensorId,
      microsensorCount: sensor.microsensors.length,

      ...this.model.process(sensor.microsensors),
    })),
  };
}
```

### 3. La serie usa `finalMean`

Dentro de `Chart.tsx`, transformo:

```ts
ProcessedBatch[]
```

en una serie temporal por sensor:

```ts
type SensorPoint = {
  batchId: number;
  receivedAt: number;
  value: number;
};

type SensorSeries = {
  sensorId: string;
  points: SensorPoint[];
};
```

La función central es:

```ts
const buildSeries = (
  batches: ProcessedBatch[],
): SensorSeries[] => {
  const map = new Map<string, SensorPoint[]>();

  batches.forEach((batch) => {
    batch.sensors.forEach((sensor) => {
      const points =
        map.get(sensor.sensorId) ?? [];

      points.push({
        batchId: batch.batchId,
        receivedAt: batch.receivedAt,

        // Resultado limpio del modelo estadístico
        value: sensor.finalMean,
      });

      map.set(sensor.sensorId, points);
    });
  });

  return [...map.entries()]
    .map(([sensorId, points]) => ({
      sensorId,
      points,
    }))
    .sort((a, b) =>
      a.sensorId.localeCompare(b.sensorId)
    );
};
```

Así, si recibes:

```text
Batch 1
Y1001 → 2.014
Y1002 → 4.831

Batch 2
Y1001 → 2.120
Y1002 → 4.794

Batch 3
Y1001 → 2.087
Y1002 → 4.910
```

Chart construye conceptualmente:

```ts
Y1001 = [2.014, 2.120, 2.087]

Y1002 = [4.831, 4.794, 4.910]
```

pero conservando también el timestamp y el batch correspondiente.

### 4. Ventana deslizante

Para un stream continuo no conviene intentar renderizar 20.000 batches simultáneamente.

Configuré:

```ts
const DEFAULT_WINDOW_SIZE = 48;
```

Y:

```ts
const visibleBatches = useMemo(
  () => batches.slice(-windowSize),
  [batches, windowSize],
);
```

Por lo tanto, el procesamiento histórico puede seguir reteniendo más información, pero la gráfica solo pinta los últimos **48 batches**.

Puedes cambiarlo fácilmente:

```tsx
<Chart
  batches={processedBatches}
  windowSize={100}
/>
```

### 5. Actualización fluida con `requestAnimationFrame`

No hice que la línea simplemente “salte” cuando llega un batch.

Cada actualización interpola desde el frame actualmente dibujado hasta el nuevo:

```ts
const ANIMATION_MS = 280;
```

```ts
const render = (timestamp: number) => {
  const linearProgress = Math.min(
    (timestamp - startedAt) / ANIMATION_MS,
    1,
  );

  const progress =
    1 - Math.pow(1 - linearProgress, 3);

  const frame = interpolateFrame(
    from,
    targetFrame,
    progress,
  );

  renderedFrameRef.current = frame;

  drawChart(
    canvas,
    frame,
    visibleBatches,
    yDomain.min,
    yDomain.max,
  );

  if (linearProgress < 1) {
    animationRef.current =
      requestAnimationFrame(render);
  }
};
```

Ese:

```ts
1 - Math.pow(1 - linearProgress, 3)
```

es un pequeño **ease-out**, por lo que el movimiento entra rápidamente y termina suavemente.

### 6. Utilicé Canvas para el stream

En esta versión preferí:

```tsx
<canvas
  ref={canvasRef}
  className="stream-chart-canvas"
/>
```

en vez de acumular muchos nodos SVG.

Para un dashboard que puede recibir datos continuamente tiene varias ventajas:

* pocos nodos en el DOM;
* redraw muy barato;
* varias líneas simultáneas;
* animación con `requestAnimationFrame`;
* no agregamos una dependencia de charts;
* permite manejar streams largos mejor;
* responde bien al redimensionar tu split Test/Chart.

Además utilizo:

```ts
const dpr =
  Math.max(window.devicePixelRatio || 1, 1);
```

para que la gráfica permanezca nítida en pantallas Retina.

Y:

```ts
const observer =
  new ResizeObserver(() => {
    drawChart(...);
  });

observer.observe(canvas);
```

por lo que si arrastras el separador entre paneles, la gráfica se reajusta automáticamente.

### 7. Sensores dinámicos

No asumí una cantidad fija de sensores.

Chart obtiene los IDs disponibles:

```ts
const sensorIds = useMemo(
  () =>
    allSeries.map(
      (series) => series.sensorId,
    ),
  [allSeries],
);
```

y genera automáticamente:

```tsx
<select
  value={selectedSensor}
  onChange={(event) =>
    setSelectedSensor(event.target.value)
  }
>
  <option value="all">
    All sensors
  </option>

  {sensorIds.map((sensorId) => (
    <option
      value={sensorId}
      key={sensorId}
    >
      {sensorId}
    </option>
  ))}
</select>
```

Por tanto puedes ver:

```text
Sensor
┌──────────────────┐
│ All sensors    ▼ │
├──────────────────┤
│ Y1001            │
│ Y1002            │
│ L1001            │
│ ...              │
└──────────────────┘
```

Si seleccionas uno, la escala Y también se recalcula para ese sensor.

### 8. KPIs de Chart

También conecté las métricas a datos reales:

```tsx
Latest avg
2.428

Window avg
2.391

Range
1.974–2.803

Batches
48
```

Cuando está seleccionado `All sensors`, `Latest avg` representa el promedio de los resultados finales del último batch:

```ts
const latestValues =
  latestBatch
    ? latestBatch.sensors
        .filter(
          (sensor) =>
            selectedSensor === 'all' ||
            sensor.sensorId === selectedSensor,
        )
        .map(
          (sensor) => sensor.finalMean,
        )
    : [];

const latest =
  latestValues.length > 0
    ? mean(latestValues)
    : 0;
```

Si seleccionas `Y1001`, entonces muestra únicamente el último `finalMean` de `Y1001`.

El resultado final es una pantalla **Chart realmente conectada al stream**, no una gráfica con datos de ejemplo: cada nuevo batch entra al parser, pasa por el modelo de tolerancia que construimos antes, y su `finalMean` aparece automáticamente en la serie con una transición de ~280 ms.

También pasé el proyecto completo por `tsc --noEmit` y **la integración actual no presenta errores de TypeScript**.
